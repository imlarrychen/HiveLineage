CREATE TEMPORARY MACRO CUST_AGE_MC(opn_dt STRING,bth_dt STRING)
    IF(DATEDIFF(opn_dt,CONCAT(SUBSTRING(opn_dt,0,4),SUBSTRING(bth_dt,5,7))) >= 0,
        (SUBSTRING(opn_dt,0,4)-SUBSTRING(bth_dt,0,4)),
        (SUBSTRING(opn_dt,0,4)-SUBSTRING(bth_dt,0,4)-1))
;

INSERT OVERWRITE TABLE mtr_cpr_100.LW68_DEE_EAC_VRF_T_H PARTITION (BIZ_DATE)
SELECT
    cust_id
    ,iac_id
    ,vrf_rst_cd
    ,dw_upd_tm
    ,ETL_HQT.bdpdatefunc('${work_day}',1,'yyyyMMdd',-1) AS BIZ_DATE
FROM (
    SELECT cust_id, iac_id, vrf_rst_cd, dw_upd_tm,
           ROW_NUMBER() OVER(PARTITION BY cust_id,iac_id ORDER BY dw_upd_tm DESC) AS RN
    FROM (
        SELECT vrf_clt_nbr AS cust_id, vrf_iac_nbr AS iac_id,
               vrf_chk_rlt AS vrf_rst_cd, CURRENT_TIMESTAMP() AS dw_upd_tm
        FROM mtr_cpr_100.LW68_DEE_EAC_VRF_T T1
        WHERE BIZ_DATE = ETL_HQT.bdpdatefunc('${work_day}',1,'yyyyMMdd',-1)
        UNION ALL
        SELECT cust_id, iac_id, vrf_rst_cd, dw_upd_tm
        FROM mtr_cpr_100.LW68_DEE_EAC_VRF_T_H
        WHERE BIZ_DATE = ETL_HQT.bdpdatefunc('${work_day}',1,'yyyyMMdd',-2)
    ) T
) T0
WHERE T0.RN = 1
;

WITH TMP0 AS (
    SELECT B1.PRCP_CUST_ID
        ,B1.PRCP_EAC_SEQ_NBR
        ,B1.CMSN_BUS_TYP_CD
        ,B1.ATN_NM
        ,B1.ATN_CNR_CD
        ,B1.ATN_DOC_NBR
        ,B1.CMSN_REL
        ,B2.RSK_GRD_CD
        ,B3.AGTH_PASS_IND AS CMSN_FACE_CMP
        ,B4.CZID AS CMSN_CZID
        ,B4.CZRESULT AS CMSN_CZRESULT
    FROM (
        SELECT AGT_CLT_NBR AS PRCP_CUST_ID
            ,AGT_IAC_NBR AS PRCP_EAC_SEQ_NBR
            ,AGT_TRX_TYP AS CMSN_BUS_TYP_CD
            ,AGT_AGT_NAM AS ATN_NM
            ,AGT_AGT_COD AS ATN_CNR_CD
            ,AGT_CTF_IDC AS ATN_DOC_NBR
            ,AGT_AGT_REL AS CMSN_REL
            ,AGT_MNT_DAT AS DW_DAT_DT
            ,ROW_NUMBER() OVER(PARTITION BY AGT_CLT_NBR,AGT_IAC_NBR ORDER BY AGT_MNT_DAT DESC) AS RN
        FROM mtr_cpr_100.LW66_DEC_AGENT_T
        WHERE AGT_TRX_TYP IN ('DCFTUOPN','DPOPNBKP','DPOPNBKF','DCOAOOPN','DCOCOOPN','IDCOAOOS')
    ) B1
    LEFT JOIN (
        SELECT DOC_NBR, DOC_TYP_CD, RSK_GRD_CD
        FROM (
            SELECT DOC_NBR, DOC_TYP_CD, RSK_GRD_CD,
                   ROW_NUMBER() OVER(PARTITION BY DOC_NBR,DOC_TYP_CD ORDER BY DW_UPD_DT,DW_UPD_TM DESC) RN
            FROM (
                SELECT DOC_NBR, DOC_TYP_CD, MAX(RSK_GRD_CD) RSK_GRD_CD, DW_UPD_DT, DW_UPD_TM
                FROM mtr_cpr_100.CORE_IDV_CUST_INF_S
                WHERE BIZ_DATE=ETL_HQT.bdpdatefunc('${work_day}',1,'yyyyMMdd',-1)
                GROUP BY DOC_NBR,DOC_TYP_CD,DW_UPD_DT,DW_UPD_TM
            ) C0
        ) C1 WHERE C1.RN=1
    ) B2 ON B1.ATN_DOC_NBR=B2.DOC_NBR
    LEFT JOIN (
        SELECT CTFIDC AS DOC_NBR, CTFTYP AS DOC_TYP, CMPRLT AS AGTH_PASS_IND,
               ROW_NUMBER() OVER(PARTITION BY CTFIDC,CTFTYP ORDER BY WRKDAT,OPTIME DESC) RN
        FROM mtr_cpr_100.LT14_T_FACECMP_H
        WHERE BIZ_DATE=ETL_HQT.bdpdatefunc('${work_day}',1,'yyyyMMdd',-1)
    ) B3 ON B1.ATN_DOC_NBR=B3.DOC_NBR
    LEFT JOIN (
        SELECT CZID,
               CASE WHEN CZRESULT='05' THEN 'NN' WHEN CZRESULT='06' THEN 'SS' ELSE CZRESULT END AS CZRESULT
        FROM (
            SELECT CZID, CZRESULT,
                   ROW_NUMBER() OVER(PARTITION BY CZID ORDER BY CZRESULT) RN
            FROM mtr_cpr_100.SLC54_T_USEROPLOG_H
            WHERE CZRESULT !='' AND CZRESULT IS NOT NULL
              AND BIZ_DATE=ETL_HQT.bdpdatefunc('${work_day}',1,'yyyyMMdd',-1)
        ) E1 WHERE E1.RN=1
    ) B4 ON B1.ATN_DOC_NBR=B4.CZID
    WHERE B1.RN = 1
),
TMP1 AS (
    SELECT EAC_EAC_NBR AS EACEACSHR
        ,EAC_KEY_SEQ AS PRE_EAC_SEQ_NBR
        ,EAC_CTY_COD AS EAC_CTY_COD
        ,EAC_BUS_MAP AS EAC_BUS_MAP
        ,EAC_FLG_MAP AS EAC_FLG_MAP
    FROM mtr_cpr_100.LW68_DEE_EAC_INF_T
    WHERE BIZ_DATE=ETL_HQT.bdpdatefunc('${work_day}',1,'yyyyMMdd',-1)
),
TMP2 AS (
    SELECT A1.EAC_ID
        ,A1.OPN_DOC_NBR
        ,A1.OPN_DOC_TYP_CD
        ,A2.CZID
        ,A2.CZRESULT
        ,A3.AGTH_PASS_IND
        ,A3.EAC_ID AS EAC_ID_2
    FROM mtr_cpr_100.MID_EAC_ACT_INF A1
    LEFT JOIN (
        SELECT CZID, WRKDAT, SBHCOD, OPUSER,
               CASE WHEN CZRESULT='05' THEN 'NN' WHEN CZRESULT='06' THEN 'SS' ELSE CZRESULT END AS CZRESULT
        FROM (
            SELECT CZID, WRKDAT, SBHCOD, OPUSER, CZRESULT,
                   ROW_NUMBER() OVER(PARTITION BY CZID,WRKDAT,SBHCOD,OPUSER ORDER BY CZRESULT) RN
            FROM mtr_cpr_100.SLC54_T_USEROPLOG_H
            WHERE BIZ_DATE=ETL_HQT.bdpdatefunc('${work_day}',1,'yyyyMMdd',-1)
              AND CZRESULT !='' AND CZRESULT IS NOT NULL
        ) B1 WHERE B1.RN=1
    ) A2 ON A1.OPN_DOC_NBR = A2.CZID
        AND A1.OPN_DT = A2.WRKDAT
        AND A1.OPN_ORG_ID = A2.SBHCOD
        AND A1.OPN_USR_ID = A2.OPUSER
    LEFT JOIN (
        SELECT CTFIDC AS DOC_NBR, CTFTYP AS DOC_TYP, CMPRLT AS AGTH_PASS_IND,
               EACNBR AS EAC_ID, WRKDAT AS TRX_DT, SBHCOD AS BRN_ORG_ID, OPUSER AS HDL_USR_ID,
               ROW_NUMBER() OVER(PARTITION BY WRKDAT,CTFIDC,SBHCOD,CTFTYP,OPUSER ORDER BY WRKDAT,OPTIME DESC) RN
        FROM mtr_cpr_100.LT14_T_FACECMP_H
        WHERE BIZ_DATE=ETL_HQT.bdpdatefunc('${work_day}',1,'yyyyMMdd',-1)
    ) A3 ON A1.OPN_DT = A3.TRX_DT
        AND A1.OPN_DOC_TYP_CD = A3.DOC_TYP
        AND A1.OPN_DOC_NBR = A3.DOC_NBR
        AND A1.OPN_ORG_ID = A3.BRN_ORG_ID
        AND A1.OPN_USR_ID = A3.HDL_USR_ID
    WHERE A1.BIZ_DATE=ETL_HQT.bdpdatefunc('${work_day}',1,'yyyyMMdd',-1)
      AND A1.PUB_OR_PRV_TYP_CD='P'
      AND A1.DMG_DAT_SRC IN ('CRDC_BAS_INF','NVLD_EAC_INF','RTL_CUST_EAC_INF','RTL_LOAN_AST_EAC_INF','UNT_LOAN_EAC_INF','WLTH_EAC_INF')
),
TMP_CHNL_CD AS (
    SELECT EAC_ID, JNL_TYP_CD, CHNL_CD, ORG_ID, HDL_TLR_ID
    FROM (
        SELECT A1.EAC_ID, A1.JNL_TYP_CD, A1.INI_CHNL_CD AS CHNL_CD,
               A1.REG_DEPT_ID AS ORG_ID, A1.HDL_TLR_ID,
               ROW_NUMBER() OVER(PARTITION BY A1.EAC_ID,A1.JNL_TYP_CD ORDER BY A1.REG_DT DESC,A1.REG_TM DESC,A1.SRC_TBL DESC) AS RN
        FROM (
            SELECT EAC_EAC_NBR AS EAC_ID, EAC_LOC_TYP AS JNL_TYP_CD,
                   EAC_CNL_COD AS INI_CHNL_CD, EAC_LOG_BRN AS REG_DEPT_ID,
                   EAC_OPR_TLR AS HDL_TLR_ID, EAC_LOG_DAT AS REG_DT, EAC_LOG_TIM AS REG_TM, 1 AS SRC_TBL
            FROM mtr_cpr_100.LU39_DEE_EAC_LOG_T
            WHERE BIZ_DATE=ETL_HQT.bdpdatefunc('${work_day}',1,'yyyyMMdd',-1)
            UNION ALL
            SELECT EAC_ID, JNL_TYP_CD, INI_CHNL_CD, REG_DEPT_ID,
                   HDL_TLR_ID, REG_DT, REG_TM, 0 AS SRC_TBL
            FROM mtr_cpr_100.ST05_EAC_STS_JNL_INF_H
            WHERE BIZ_DATE=ETL_HQT.bdpdatefunc('${work_day}',1,'yyyyMMdd',-2)
        ) A1
    ) A1 WHERE A1.RN = 1
),
TMP_SSP_INF AS (
    SELECT A1.CUST_ID, A1.EAC_SEQ_NBR,
           CASE WHEN A1.TMN_ALL_BUS_IND ='Y' THEN '1' ELSE '0' END AS SSP_ALL_BUS_IND
    FROM mtr_cpr_100.MID_CORE_CUST_EAC_ATCH_INF A1
    LEFT SEMI JOIN mtr_cpr_100.MID_CORE_CUST_EAC_INF_S A2
        ON A1.EAC_ID=A2.EAC_ID AND A2.SEQ_NO='00'
       AND A2.BIZ_DATE=ETL_HQT.bdpdatefunc('${work_day}',1,'yyyyMMdd',-1)
    WHERE A1.BIZ_DATE=ETL_HQT.bdpdatefunc('${work_day}',1,'yyyyMMdd',-1)
)
